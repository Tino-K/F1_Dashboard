CREATE DATABASE f1_dashboard_db;

CREATE TABLE circuits (
    id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(50),
    lengthKM FLOAT,
    firstGP INT(10),
    numberOfLaps INT(10),
    raceDistance FLOAT,
    circuitMapUrl VARCHAR(500),
    country VARCHAR(100)
);

CREATE TABLE teams (
    id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(255),
    logo VARCHAR(500),
    car_image VARCHAR(500),
    color VARCHAR(50)
);

CREATE TABLE drivers (
    id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(100),
    surname VARCHAR(100),
    shortName VARCHAR(10),
    number INT(11),
    image VARCHAR(500),
    team_id INT,
    FOREIGN KEY (team_id) REFERENCES teams(id)
);

CREATE TABLE users (
    id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(255),
    email VARCHAR(255),
    password VARCHAR(255),
    gender VARCHAR(20),
    bio TEXT,
    avatar_url VARCHAR(500),
    role ENUM('user', 'admin') DEFAULT 'user',
    theme_preference ENUM('light', 'dark') DEFAULT 'light',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE requests (
    id INT PRIMARY KEY AUTO_INCREMENT,
    Guest TINYINT(1),
    Text TEXT,
    Email VARCHAR(100),
    taken BOOLEAN DEFAULT FALSE,
    name TEXT
);
